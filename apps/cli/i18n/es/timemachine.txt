# Mensajes de `raptor` para el usuario, en español (NFR-10, ADR-GRP-003): timemachine.
# Una línea `clave = texto`. Los marcadores son `{nombre}`; el código los
# rellena por nombre, nunca concatenando. Toda clave existe también en ../en/timemachine.txt.

# El snapshot previo falló: la operación no se ejecutó (US-TMC-001).
prior.no-space = no se ejecutó: no hay espacio en disco para guardar antes el trabajo; nada cambió
prior.store-unavailable = no se ejecutó: el almacén de snapshots no está disponible; nada cambió
prior.timeout = no se ejecutó: guardar antes el trabajo tardó demasiado; nada cambió
prior.daemon-stopping = no se ejecutó: el motor se está deteniendo; nada cambió
prior.capture-failed = no se ejecutó: no se pudo guardar antes el trabajo; nada cambió

# raptor undo (US-TMC-002).
undo.done = deshecha {operation} ({id}) en {worktree}
undo.saved = el estado justo antes de este undo queda guardado como {snapshot}; no se perdió nada
undo.not-restored = se dejó como estaba ({reason}): {path}
undo.path.overlap = alguien más lo cambió mientras tanto
undo.path.not-guaranteed = este sistema de archivos no permite restaurarlo con garantía
undo.path.blocked = una carpeta del camino es un enlace o está en otro disco
undo.operation-unnamed = la operación
undo.not-in-worktree = no estás dentro de un worktree de Git
undo.not-observed = {worktree} no está en un repo que GitRaptor observe; añádelo con raptor repo add
undo.interrupted = interrumpido a mitad ({id}); ejecuta raptor undo otra vez para volver al estado anterior
undo.reason.nothing-to-undo = no hay nada que deshacer en este worktree; nada cambió
undo.reason.raw-git-not-covered = el último cambio se hizo con Git directamente y deshacer ese tipo de cambio aún no está disponible; nada cambió
undo.reason.target-unavailable = el estado previo a esa operación ya no está disponible; nada cambió
undo.reason.other-actor = ese trabajo es de otro actor; nada cambió
undo.reason.confirmation-required = ese trabajo es de un agente: pide al agente que ejecute él mismo raptor undo (confirmarlo desde tu terminal aún no está disponible); nada cambió
undo.reason.git-operation-in-progress = hay una operación de Git en curso (merge, rebase, cherry-pick...); termínala o abórtala antes; nada cambió
undo.reason.git-busy = Git está ocupado (hay un archivo de lock); nada cambió
undo.reason.repo-busy = hay otra operación en curso en este repo; inténtalo de nuevo; nada cambió
undo.reason.repo-untrusted = Git no confía en este repo (safe.directory); nada cambió
undo.reason.worktree-unavailable = falta o se movió un worktree de esa operación; nada cambió
undo.reason.invalid-snapshot = un estado guardado no pasó su verificación de integridad; nada cambió
undo.reason.hostile-tree = el estado guardado tiene rutas inseguras en este sistema de archivos; nada cambió
undo.reason.ref-moved = una rama se movió mientras tanto; nada cambió
undo.reason.ref-in-use = una rama que movería está en uso en otro worktree; nada cambió
undo.reason.unsupported = el undo todavía no está soportado aquí; nada cambió
undo.reason.git-unavailable = el motor no encontró un Git utilizable; nada cambió

# raptor timeline (US-TMC-006).
timeline.line = {time}  {worktree}  {what}  ({actor})  [{protection}]
timeline.empty = aún no hay actividad
timeline.no-match = ninguna entrada coincide con los filtros
timeline.incomplete = timeline incompleto: no se pudo leer {sources}
timeline.truncated = no se muestran las entradas más antiguas (acota con --since o sube --limit)
timeline.source.operations = las operaciones de la Time Machine
timeline.source.events = los eventos de Git
timeline.actor-unavailable = agente no disponible
timeline.level.prior = snapshot previo
timeline.level.observation = capturado por observación
timeline.level.none = sin protección
timeline.op.protected = operación protegida: {subtype}
timeline.op.protected-unnamed = operación protegida
timeline.op.undo = undo
timeline.op.redo = redo
timeline.op.restore = restauración
timeline.state.applying = en curso
timeline.state.interrupted = interrumpida
timeline.branch-unknown = rama desconocida
timeline.acted-on = actuó sobre: {targets}
timeline.acted.operation = operación {id}
timeline.acted.event = evento {seq}
timeline.acted.snapshot = snapshot {id}
timeline.files = archivos ({total}): {paths}
timeline.files-more = +{count} más
timeline.files-none = archivos: ninguno
timeline.files-unavailable = archivos: no disponible
timeline.files-first-parent = respecto al primer padre
timeline.not-in-worktree = no estás dentro de un worktree de Git
timeline.not-observed = {worktree} no está en un repo que GitRaptor observe; añádelo con raptor repo add
timeline.invalid-params = los filtros no son válidos (--since es una duración como 2h, hasta 30 días; --limit va de 1 a 200)
timeline.restart-engine = el motor en marcha es anterior a este raptor y no puede mostrar el timeline: reinícialo con `raptor daemon stop`
