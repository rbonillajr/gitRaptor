# Guardrails (US-GRD-001). Los mensajes de `raptor hook` los lee el agente: una plantilla fija por
# código y parámetros etiquetados «…»; nunca una excepción ni cómo quitar la protección (M-05).
guard.deny.force-push = GitRaptor: force-push a {branch} en {remote} denegado. Regla «prohibir force-push» del conjunto mínimo por defecto.
guard.deny.shallow = GitRaptor: push a {branch} denegado. El clon es superficial y no se puede comprobar que no reescribe historia (regla «prohibir force-push»). Si es legítimo, ejecuta git fetch --unshallow y reintenta.
guard.deny.base-delete = GitRaptor: borrar {branch} denegado. Protección de la rama base.
guard.deny.alias = GitRaptor: operación sobre {ref} denegada: ese nombre equivale a la rama base {base}. Protección de la rama base.
guard.deny.rename-onto-base = GitRaptor: renombrar sobre la rama base {base} denegado. Protección de la rama base. Git ya borró {branch}; su último commit era {oid}.
guard.deny.channel = GitRaptor: no se pudo verificar el servicio de GitRaptor. La operación no se ejecuta.
guard.deny.repo-mismatch = GitRaptor: esta operación no es del repositorio que GitRaptor protege aquí. La operación no se ejecuta.
guard.deny.input = GitRaptor: no se pudo leer la entrada del hook. La operación no se ejecuta.
guard.deny.internal = GitRaptor: error interno al decidir. La operación no se ejecuta.
guard.degraded = GitRaptor: sin conexión con el servicio ({reason}). Se aplica el conjunto mínimo con su criterio más estricto.
guard.degraded.unreachable = el servicio no está en marcha
guard.degraded.instance = respondió otro perfil de GitRaptor
guard.refused-agent = rechazado: solo el desarrollador puede cambiar la protección de un repo, no un agente ni un proceso lanzado por él
guard.unsupported-platform = todavía no disponible en esta plataforma (falta el canal del motor en Windows)
guard.install.header = Proteger {repo} con los hooks de GitRaptor
guard.install.what = Qué: los hooks {hooks} en {dir}, y un ajuste local de Git (core.hooksPath) que apunta a ellos.
guard.install.where = Dónde: solo en este repositorio ({repo}). Nada en tu configuración global de Git ni en otros repositorios.
guard.install.scope = Cubre todos los worktrees de este repositorio ({count} hoy, y los que se creen después).
guard.install.why = Por qué: para que ningún agente haga force-push ni borre la rama base, aunque use Git directo.
guard.install.base = Rama base confirmada para este repositorio: {base}.
guard.install.base-unconfirmed = Este repositorio tiene configuración del equipo: la rama base queda sin confirmar y se protegen estas ramas: {bases}.
guard.install.cannot-prevent = Lo que los hooks no pueden impedir en este repositorio:
guard.install.reftable = Este repositorio usa reftable: renombrar la rama base no ejecuta ningún hook.
guard.install.cost = Coste: cada operación de Git ejecuta un hook nativo pequeño; las operaciones masivas (fetch o gc con miles de refs) tardan más, en proporción a las refs.
guard.install.unavailable = Si GitRaptor falta o falla, push y rebase quedan bloqueados hasta que lo restaures; los commits siguen funcionando.
guard.install.revert = Para revertirlo a mano:
guard.install.revert-folder =   y después borra la carpeta {dir}
guard.install.if-not = Si no lo autorizas, no cambia nada y el repositorio sigue sin protección.
guard.install.confirm = ¿Proteger este repositorio? [s/N]
guard.install.no-answer = sin respuesta: no se instaló nada
guard.install.declined = no se instaló; GitRaptor no volverá a ofrecerlo para este repositorio (ejecuta raptor guard install para hacerlo tú)
guard.install.done = protegido: {path}
guard.install.already = ya estaba protegido: {path}
guard.install.refused = no se instaló:
guard.state.unprotected = estado: sin protección
guard.state.hooks-only = estado: solo hooks
guard.status.base-confirmed = rama base protegida: {bases} (confirmada)

guard.permission.not-asked = permiso: no pedido
guard.permission.granted = permiso: concedido
guard.permission.denied = permiso: denegado (no se vuelve a ofrecer)
guard.np.reset-hard = git reset --hard (la Time Machine puede deshacerlo)
guard.np.merge = merge, también fast-forward
guard.np.remove-worktree = borrar un worktree
