# Mensajes de `raptor` para el usuario, en español (NFR-10, ADR-GRP-003): events.
# Una línea `clave = texto`. Los marcadores son `{nombre}`; el código los
# rellena por nombre, nunca concatenando. Toda clave existe también en ../en/events.txt.

kind.added = añadido
kind.deleted = borrado
kind.modified = modificado
kind.type-changed = cambio de tipo
kind.conflicted = en conflicto

events.none = todavía no hay eventos de Git
events.line = {time}  {worktree}  {event}  ({actor})
events.inferred = {worktree} (deducido)
events.restart-engine = el motor en marcha es anterior a este raptor y no puede mostrar eventos: reinícialo con `raptor daemon stop`
events.commit_by = commit de {author}
events.merge_by = merge de {author}
events.commit_with = con {agents}
events.run_by = ejecutado por {agent}
events.no_trailer = sin trailer
events.inferred_confirmed = confirmado por el trailer
events.inferred_unconfirmed = no confirmado por el trailer
events.no_agent = sin agente
events.no_agent_inferred = sin agente; inferido: {agent}

event.commit = commit en {branch}
event.merge = merge en {branch}
event.rebase = rebase de {branch}
event.branch-update = la rama {branch} se movió
event.branch-create = rama {branch} creada
event.branch-delete = rama {branch} borrada
event.branch-switch = cambio de la rama {from} a {branch}
event.branch-switch-to = cambio a la rama {branch}
event.worktree-create = worktree creado en {branch}
event.worktree-delete = worktree borrado
event.push = push a {branch}
event.reconciled = reconciliación: cambios encontrados tras un hueco (no es un comando de Git)
event.reset = reset en {branch} sin moverla
event.reset-detached = reset con HEAD separado

events.gaps-restart-engine = el motor en marcha es anterior a este raptor y no puede listar los huecos de observación: reinícialo con `raptor daemon stop`
events.gap_line = -- hueco de observación ({cause}), {from} → {to} · sin atribuir
events.gap_open = -- hueco de observación ({cause}), desde las {from} (abierto) · sin atribuir
events.gap_during_session = había una sesión de agente activa

gap.cause.machine-off = equipo apagado
gap.cause.daemon-down = motor detenido
gap.cause.daemon-down-during-session = motor detenido
gap.cause.daemon-stopped = motor detenido
gap.cause.repo-retired = repo retirado
gap.cause.git-unavailable = Git no disponible
gap.cause.profile-lost = perfil perdido
gap.cause.store-corrupt = almacén dañado
gap.cause.watcher-overflow = el vigilante de cambios se desbordó
gap.cause.stream-recreated = flujo de cambios recreado
gap.cause.periodic-reconciliation = revisión periódica
gap.cause.dormant = repo inactivo
