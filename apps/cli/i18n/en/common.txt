# User-facing messages of `raptor`, English (NFR-10, ADR-GRP-003): common.
# One `key = text` per line. Placeholders are `{name}`; the code fills them
# by name, never by concatenating. Every key exists in ../es/common.txt too.

common.not-running = not running
common.cancelled = cancelled
common.confirm-needs-terminal = confirmation needs a terminal (or pass --yes)
common.refused-terminal = refused: run it from your own terminal
common.refused-terminal-no-console = refused: this process has no console; run it from a console window (PowerShell, cmd or Windows Terminal)
common.refused-terminal-session = refused: not from an SSH session or a service (session 0); run it from a local or Remote Desktop console
common.refused-terminal-foreign-console = refused: this console was not opened by your shell (another process attached to it); run it from a console you opened
common.refused-unverified = refused: the engine could not verify who is asking
common.refused-unverified-console = refused: reserved commands are only accepted from an interactive console (PowerShell, cmd or Windows Terminal); this terminal does not let the engine verify who is asking
common.refused-unverified-terminal = refused: reserved commands are only accepted from an interactive terminal; this one does not let the engine verify who is asking
